% Minimal preamble for building individual chapters standalone during drafting.
% The real thesis will have its own; this exists so a chapter can be checked without one.

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}

% without texlive-science. The real thesis preamble should require siunitx properly.

  \providecommand{\SI}[2]{##1\,##2}%
  \providecommand{\num}[1]{##1}%
  \providecommand{\SIrange}[3]{##1--##2\,##3}%
  \providecommand{\percent}{\%}%
  \providecommand{\centi}{c}\providecommand{\milli}{m}\providecommand{\kilo}{k}%
  \providecommand{\metre}{m}\providecommand{\second}{s}\providecommand{\hour}{h}%
  \providecommand{\gram}{g}\providecommand{\hertz}{Hz}\providecommand{\newton}{N}%
  \providecommand{\degree}{$^\circ$}\providecommand{\per}{/}\providecommand{\squared}{$^2$}%
}
